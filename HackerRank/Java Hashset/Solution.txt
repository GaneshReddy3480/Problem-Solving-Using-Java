import java.io.*;
import java.util.*;

public class Solution {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int t = s.nextInt();
        String [] pair_left = new String[t];
        String [] pair_right = new String[t];
        
        for (int i = 0; i < t; i++) {
            pair_left[i] = s.next();
            pair_right[i] = s.next();
        }

        // Write your code here
        // 1. Initialize the HashSet to the maximum size to prevent resizing delays
        HashSet<String> set = new HashSet<>(t);
        
        // 2. Iterate and append using a delimiter so ("a", "bc") doesn't match ("ab", "c")
        for (int i = 0; i < t; i++) {
            set.add(pair_left[i] + " " + pair_right[i]);
            System.out.println(set.size());
        }
        
        s.close();
    }
}
